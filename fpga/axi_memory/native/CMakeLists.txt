cmake_minimum_required(VERSION 3.16)
project(shoop_dram_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(THERMAL OFF CACHE BOOL "" FORCE)
set(CMD_TRACE OFF CACHE BOOL "" FORCE)
set(ADDR_TRACE OFF CACHE BOOL "" FORCE)
add_subdirectory("${DRAMSIM3_SOURCE}" vendor)
set_target_properties(dramsim3 PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
add_library(shoop_dram SHARED bridge.cpp)
target_link_libraries(shoop_dram PRIVATE dramsim3 inih format json)
target_include_directories(shoop_dram SYSTEM PRIVATE "${DRAMSIM3_SOURCE}/src" "${DRAMSIM3_SOURCE}/ext/headers" "${DRAMSIM3_SOURCE}/ext/fmt/include")
target_compile_options(shoop_dram PRIVATE -Wall -Wextra -Werror)
set_target_properties(shoop_dram PROPERTIES BUILD_RPATH "$ORIGIN")
